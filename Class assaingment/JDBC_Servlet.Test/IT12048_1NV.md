Task 02 --- JDBC: Find the nth Prime Number and Store It in MySQL
Objective
Take a positive integer n from the user, calculate the nth prime
number, and insert both n and the prime number into a MySQL table
using JDBC.
Open Beekeper Stdio
Enter the MySQL root password when prompted. Then execute:
CREATE DATABASE integer_db;

USE integer_db;

CREATE TABLE integers (
    id INT AUTO_INCREMENT PRIMARY KEY,
    number INT NOT NULL
);
 Java source code
File: [PrimeJDBCTest/src/main/java/org/example/Main.java](https://github.com/Sabbir-byte01/Software-Engineering/blob/main/Class%20assaingment/JDBC_Servlet.Test/IntegerDatabase.java)
import java.sql.*;
import java.util.Scanner;

public class IntegerDatabase {

    public static boolean isPrime(int num) {

        if (num < 2) {
            return false;
        }

        for (int i = 2; i <= Math.sqrt(num); i++) {
            if (num % i == 0) {
                return false;
            }
        }

        return true;
    }

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter the prime number position: ");
        int n = sc.nextInt();

        if (n <= 0) {
            System.out.println("Please enter a positive number.");
            sc.close();
            return;
        }

        String url = "jdbc:mysql://localhost:3306/integer_db";
        String user = "root";
        String password = "12345";

        String sql = "INSERT INTO integers (number) VALUES (?)";

        try (Connection con = DriverManager.getConnection(url, user, password);
             PreparedStatement pst = con.prepareStatement(sql)) {

            int count = 0;
            int num = 2;
            int primeNumber = 0;

            while (count < n) {

                if (isPrime(num)) {
                    count++;
                    primeNumber = num;
                }

                num++;
            }

            pst.setInt(1, primeNumber);
            pst.executeUpdate();

            System.out.println("The " + n + "th prime number is: " + primeNumber);
            System.out.println("Data saved successfully!");

        } catch (SQLException e) {
            e.printStackTrace();
        }

        sc.close();
    }
}
}

Task 03 --- Hello World Servlet Using Apache Tomcat
Objective
Create a Java Servlet that responds to an HTTP GET request and displays
a Hello World message in the browser.
 HelloServlet source code
File: [ServletTasks/src/main/java/org/example/HelloServlet.java](https://github.com/Sabbir-byte01/Software-Engineering/blob/main/Class%20assaingment/JDBC_Servlet.Test/HelloWorldServlet..java)

package com.example;

import jakarta.servlet.ServletException;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;

import java.io.IOException;
import java.io.PrintWriter;

@WebServlet("/hello")
public class HelloWorldServlet extends HttpServlet {

    @Override
    protected void doGet(HttpServletRequest request,
                         HttpServletResponse response)
            throws ServletException, IOException {

        response.setContentType("text/html;charset=UTF-8");

        PrintWriter out = response.getWriter();

        out.println("<html>");
        out.println("<head><title>Hello World</title></head>");
        out.println("<body>");
        out.println("<h1>Hello World!</h1>");
        out.println("<h3>Welcome to My First Web Application</h3>");
        out.println("</body>");
        out.println("</html>");
    }
}
